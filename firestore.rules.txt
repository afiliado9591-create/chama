rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function banned(uid) {
      return exists(/databases/$(database)/documents/bannedUsers/$(uid));
    }

    function activeUser() {
      return signedIn() &&
        !banned(request.auth.uid) &&
        (!exists(/databases/$(database)/documents/users/$(request.auth.uid)) ||
          !(('suspended' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data) &&
            get(/databases/$(database)/documents/users/$(request.auth.uid)).data.suspended == true));
    }

    function isAdmin() {
      return signedIn() &&
        !banned(request.auth.uid) &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        ('admin' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.admin == true;
    }

    function userBlocked(userId, otherId) {
      return exists(/databases/$(database)/documents/users/$(userId)) &&
        ('bloqueados' in get(/databases/$(database)/documents/users/$(userId)).data) &&
        (otherId in get(/databases/$(database)/documents/users/$(userId)).data.bloqueados);
    }

    function blockedEitherWay(senderId, receiverId) {
      return userBlocked(senderId, receiverId) || userBlocked(receiverId, senderId);
    }

    match /users/{userId} {
      allow read: if activeUser() || isAdmin();
      allow create: if signedIn() && request.auth.uid == userId && !banned(userId) &&
        !request.resource.data.keys().hasAny(['admin','suspended','adminBanner','adminUpdatedAt']);
      allow update: if isAdmin() ||
        (activeUser() && request.auth.uid == userId &&
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['admin','suspended','adminBanner','adminUpdatedAt']));
      allow delete: if isAdmin() || (activeUser() && request.auth.uid == userId);
    }

    match /bannedUsers/{userId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /publicProfiles/{userId} {
      allow read: if true;
      allow create, update, delete: if isAdmin() || (activeUser() && request.auth.uid == userId);
    }

    match /chats/{chatId} {
      allow read: if activeUser() && request.auth.uid in resource.data.participants;
      allow create: if activeUser() && request.auth.uid in request.resource.data.participants;
      allow update: if activeUser() &&
        request.auth.uid in resource.data.participants &&
        request.resource.data.participants == resource.data.participants;

      match /messages/{messageId} {
        allow read: if activeUser() &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
        allow create: if activeUser() &&
          request.resource.data.senderId == request.auth.uid &&
          request.resource.data.receiverId is string &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants &&
          !blockedEitherWay(request.resource.data.senderId, request.resource.data.receiverId);
        allow delete: if activeUser() &&
          resource.data.senderId == request.auth.uid &&
          request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
        allow update: if false;
      }
    }

    match /calls/{callId} {
      allow create: if activeUser() &&
        request.resource.data.callerId == request.auth.uid &&
        request.resource.data.calleeId is string &&
        request.resource.data.calleeId != request.auth.uid &&
        !blockedEitherWay(request.resource.data.callerId, request.resource.data.calleeId);
      allow read: if activeUser() &&
        (request.auth.uid == resource.data.callerId || request.auth.uid == resource.data.calleeId);
      allow update: if activeUser() &&
        (request.auth.uid == resource.data.callerId || request.auth.uid == resource.data.calleeId) &&
        request.resource.data.callerId == resource.data.callerId &&
        request.resource.data.calleeId == resource.data.calleeId;
      allow delete: if activeUser() &&
        (request.auth.uid == resource.data.callerId || request.auth.uid == resource.data.calleeId);

      match /{candidateCollection}/{candidateId} {
        allow read, create, delete: if activeUser() &&
          (request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.callerId ||
           request.auth.uid == get(/databases/$(database)/documents/calls/$(callId)).data.calleeId);
        allow update: if false;
      }
    }

    match /channels/{channelId} {
      allow read: if activeUser();
      allow create: if activeUser() &&
        request.resource.data.ownerId == request.auth.uid &&
        request.auth.uid in request.resource.data.followers;
      allow update: if activeUser() &&
        (
          resource.data.ownerId == request.auth.uid ||
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['followers','updatedAt'])
        ) &&
        request.resource.data.ownerId == resource.data.ownerId;
      allow delete: if activeUser() && resource.data.ownerId == request.auth.uid;

      match /posts/{postId} {
        allow read: if activeUser();
        allow create: if activeUser() &&
          request.resource.data.ownerId == request.auth.uid &&
          get(/databases/$(database)/documents/channels/$(channelId)).data.ownerId == request.auth.uid;
        allow update, delete: if activeUser() &&
          get(/databases/$(database)/documents/channels/$(channelId)).data.ownerId == request.auth.uid;
      }
    }
  }
}
